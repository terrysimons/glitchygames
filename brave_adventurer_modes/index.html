Brave Adventurer's launcher: one command for both games, Classic and Papyrus.

main() (brave_adventurer_modes/launcher.py) starts the game --game names, else the game a recording to replay was made in, else the title screen (brave_adventurer_modes/title.py), which offers both.

Brave Adventurer's launcher: main(), the one command (brave-adventurer) for both games.

The project holds two games on shared primitives: Classic (brave_adventurer), whose sprites are drawn in code, and Papyrus (brave_adventurer_papyrus), the Egyptian edition with sprite-sheet art. main() starts one of them, in this order:

  1. --game classic|papyrus: read (parse_known_args) and removed from sys.argv, then that game's own main() runs unchanged, so its arguments, scene flow, recording header and recordings stay its own. With a recording to replay as well, the flag still decides: the game's own replay refuses another game's recording.
  2. A recording to replay (--replay-trace PATH, as the golden-replay plugin, the run ledger's rebaselines, watched replays and their restarts start one): its header names the game it was made in by logical name (glitchygames/logical_names.py), and that game's own main() runs, without the title screen (game_for_recording). So a member with one console script replays both games' goldens (glitchygames/testing/replays.py recording_console_script). A recording made from the title screen names the title screen: it is replayed from there. A recording of any other game is refused (ExitCode.UNUSABLE_INPUT, naming the games this command replays); one whose header cannot be read goes to the engine, which refuses it as it refuses any unusable or missing recording.
  3. Otherwise the title screen (brave_adventurer_modes/title.py) offers both games and starts the chosen one's scene in the same engine.

def game_for_recording(*, path: Path) -> str | None:
    """Return the name of the game a recording was made in.

    Args:
        path: The recording.

    Returns:
        ``'classic'`` or ``'papyrus'``: the game whose scene the recording's header names by
        logical name; None when the recording is not one of these games' (another game's, the
        title screen's, one made before logical names, or a file that cannot be read).

    """
    try:
        logical_game = recorded_game(path=path)
    # A TraceFormatError is a ValueError
    except OSError, ValueError:
        return None
    return game_named_by(logical_game=logical_game)

    """Return the name of the game whose scene has a logical name.

    Args:

def main() -> None:
    """Start the game ``--game`` names, else a recording's game, else the title screen."""
    launcher_options, remaining_arguments = launcher_arguments().parse_known_args(sys.argv[1:])
    # Each game's own parser (and the title screen's) reads sys.argv: without --game
    sys.argv = [sys.argv[0], *remaining_arguments]
    recording = recording_to_replay(arguments=remaining_arguments)
    if launcher_options.game is not None:
        start = GAMES_BY_NAME[launcher_options.game]
    elif recording is not None:
        start = recorded_games_start(recording=recording)
    else:
        start = start_title_screen
    start()

def recorded_game(*, path: Path) -> str | None:
    """Return the logical name a recording's header names its game by.

    A file that cannot be read raises what ``TraceReader`` raises: an OSError
    (FileNotFoundError when it does not exist), or a ValueError when it is not a recording
    (glitchygames.tracing.TraceFormatError).

    Args:
        path: The recording.

    Returns:
        The header's ``game`` (``games.brave_adventurer.game:BraveAdventurerScene``), or None
        for a recording made before logical names (it has none).

    """
    with TraceReader(path=path) as reader:
        recorded = reader.header.get(GAME_HEADER_FIELD)
    return recorded if isinstance(recorded, str) else None

def recorded_games_start(*, recording: Path) -> Callable[[], None]:
    """Return what starts the game a recording to replay was made in.

    Args:
        recording: The recording (``--replay-trace``).

    Returns:
        That game's own ``main()``; or ``start_title_screen`` for a recording made from the
        title screen, and for one whose header cannot be read (the engine refuses it with its
        own exit code and message, missing or unusable, as any game's would).

    """
    try:
        logical_game = recorded_game(path=recording)
    # A TraceFormatError is a ValueError
    except OSError, ValueError:
        return start_title_screen
    if logical_game == logical_name(scene_class=TitleScene):
        return start_title_screen
    game = game_named_by(logical_game=logical_game)
    if game is None:
        refuse_recording(recording=recording, logical_game=logical_game)
    return GAMES_BY_NAME[game]

def recording_to_replay(*, arguments: Sequence[str]) -> Path | None:
    """Return the recording the arguments ask the engine to replay, without removing it.

    Args:
        arguments: The command-line arguments (without the program).

    Returns:
        ``--replay-trace``'s path, or None.

    parser = argparse.ArgumentParser(add_help=False)

    known, _others = parser.parse_known_args(list(arguments))
    return None if known.replay_trace is None else Path(known.replay_trace)

def refuse_recording(*, recording: Path, logical_game: str | None) -> NoReturn:
    """Refuse to replay a recording none of this command's scenes made.

    Args:
        recording: The recording.
        logical_game: The game its header names, or None for a recording made before logical
            names (it names none: ``--game`` names it, and the game's replay then checks the
            recording's class and module as it always has).

    Raises:
        SystemExit: Always, with ``ExitCode.UNUSABLE_INPUT``.

    """
    if logical_game is None:
        log.error(
            'Cannot replay %s: it was made before logical names, so its header names no game '
            'to start; name the game: --game %s',
            recording,
            ' or --game '.join(GAMES_BY_NAME),
        )
    else:
        log.error(
            'Cannot replay %s: it is a recording of %s; this command replays only %s',
            recording,
            logical_game,
            ', '.join([
                *(logical_name(scene_class=choice.scene_class) for choice in GAME_CHOICES),
                logical_name(scene_class=TitleScene),
            ]),
        )
    raise SystemExit(ExitCode.UNUSABLE_INPUT)

The title screen: both Brave Adventurer games, a line each on how they differ; choose one.

The screen draws the desert from the shared primitives (brave_adventurer_core's sky and pyramids), the title, a card for each game (GAME_CHOICES) and a line of help. One card is chosen at a time.

Controls, all through the scene's ActionMap (as the games read theirs), keyboard by key position:

  • Up or Left, Down or Right (a gamepad's D-pad): choose the other game (the two wrap around).
  • Space or Return (a gamepad's A or Start): start the chosen game, on the release of a press this screen saw, as the pause and game-over screens go on (glitchygames/scenes/builtin_scenes/ gamepad_choice.py): the game then never gets the release of the key that started it.
  • Q (a gamepad's B or Back): quit, on the press. Q quits on its press in the games too, and Escape (or B or Back) quits on the pause and game-over screens.
  • Escape does nothing here, on its press or its release: the browser build's touch Pause button sends Escape, so a phone player who taps it before choosing a game must not end the run.

Starting a game sets next_scene to a new scene of that game, built with the run's options, in the same engine: the game runs as it does from its own main(), which builds that scene.

class GameChoiceCard(Sprite):
    """A game's card: its name and how it differs, framed in gold while it is the chosen one."""

    def __init__(
        self: Self,
        *,
        choice: GameChoice,
        center: tuple[int, int],
        selected: bool,
        groups: pygame.sprite.LayeredDirty[Any],
    ) -> None:
        """Draw the card.

        Args:
            choice: The game.
            center: Where its centre goes on the screen.
            selected: Whether it is the chosen card.
            groups: The scene's sprites.

        """
        width, height = CARD_SIZE
        super().__init__(
            x=center[0] - width // 2,
            y=center[1] - height // 2,
            width=width,
            height=height,
            groups=groups,
        )
        self.choice = choice
        self.selected = selected
        self._name_font = pygame.font.Font(None, CARD_NAME_FONT_SIZE)
        self._description_font = pygame.font.Font(None, CARD_DESCRIPTION_FONT_SIZE)
        self._draw()

    def select(self: Self, *, selected: bool) -> None:
        """Draw the card chosen or not.

        Args:
            selected: Whether it is the chosen card.

        """
        self.selected = selected
        self._draw()

    def _draw(self: Self) -> None:
        """Draw the panel, its border, the game's name and its line; mark the card for redrawing."""
        image = pygame.Surface(CARD_SIZE, pygame.SRCALPHA)
        panel_color, border_color, border_width, name_color = (
            (CHOSEN_PANEL_COLOR, CHOSEN_BORDER_COLOR, CHOSEN_BORDER_WIDTH, CHOSEN_NAME_COLOR)
            if self.selected
            else (OTHER_PANEL_COLOR, OTHER_BORDER_COLOR, OTHER_BORDER_WIDTH, OTHER_NAME_COLOR)
        )
        card_rect = image.get_rect()
        pygame.draw.rect(image, panel_color, card_rect, border_radius=CARD_BORDER_RADIUS)
        pygame.draw.rect(
            image, border_color, card_rect, width=border_width, border_radius=CARD_BORDER_RADIUS
        )
        name = self._name_font.render(self.choice.title, True, name_color)  # ruff: ignore[boolean-positional-value-in-call] - pygame's antialias argument
        description = self._description_font.render(
            self.choice.description,
            True,  # ruff: ignore[boolean-positional-value-in-call] - pygame's antialias argument
            DESCRIPTION_COLOR,
        )
        image.blit(name, (CARD_PADDING, CARD_PADDING))
        image.blit(
            description, (CARD_PADDING, CARD_SIZE[1] - CARD_PADDING - description.get_height())
        )
        self.image = image
        self.dirty = 1

class TitleScene(Scene):
    """The launcher's title screen: choose Classic or Papyrus.

    Attributes:
        NAME: The command's name in its help and ``--version`` (the Classic game's name, which
            the command printed before it became the launcher).
        VERSION: Its version.

    """

    NAME = 'Brave Adventurer'
    VERSION = '1.0'

    def __init__(
        self: Self,
        options: dict[str, Any] | None = None,
        groups: pygame.sprite.LayeredDirty[Any] | None = None,
    ) -> None:
        """Build the screen: the backdrop, the title, a card per game and the help line.

        Args:
            options: Command-line options dict from GameEngine.
            groups: Sprite group (created if not provided).

        """
        if groups is None:
            groups = pygame.sprite.LayeredDirty()
        super().__init__(options=options, groups=groups)
        # Named, so no part of the screen the backdrop leaves shows the test pattern
        self.background_color = SKY_COLOR
        self.actions = self._create_action_map(input_mode=self.options.get('input_mode', 'auto'))
        # The chosen game (an index into GAME_CHOICES)
        self.selected_index = 0
        # A choose control's press reached this screen, so its release starts the game
        self._choose_pressed = False

        # Drawn in this order, each over the one before (LayeredDirty keeps the order it was given)
        center_x = self.screen_width // 2
        TitleBackdrop(size=(self.screen_width, self.screen_height), groups=self.all_sprites)
        for color, offset in ((HUD_SHADOW_COLOR, TITLE_SHADOW_OFFSET), (CHOSEN_NAME_COLOR, 0)):
            self.all_sprites.add(
                TextSprite(
                    TITLE_TEXT,
                    (center_x + offset, TITLE_CENTER_Y + offset),
                    color=color,
                    font_size=TITLE_FONT_SIZE,
                )
            )
        self.all_sprites.add(
            TextSprite(
                SUBTITLE_TEXT,
                (center_x, SUBTITLE_CENTER_Y),
                color=HUD_TEXT_COLOR,
                font_size=SUBTITLE_FONT_SIZE,
            )
        )
        card_height = CARD_SIZE[1]
        self.cards = [
            GameChoiceCard(
                choice=choice,
                center=(center_x, CARD_TOP_Y + index * (card_height + CARD_GAP) + card_height // 2),
                selected=index == self.selected_index,
                groups=self.all_sprites,
            )
            for index, choice in enumerate(GAME_CHOICES)
        ]
        self.all_sprites.add(
            TextSprite(
                HELP_TEXT,
                (center_x, self.screen_height - HELP_BOTTOM_MARGIN),
                color=HUD_TEXT_COLOR,
                font_size=HELP_FONT_SIZE,
            )
        )

    @staticmethod
    def _create_action_map(*, input_mode: str) -> ActionMap:
        """Bind the controls: keys by position, and a gamepad's D-pad, A, Start, B and Back.

        Args:
            input_mode: Gamepad event family ('joystick', 'controller' or 'auto').

        Returns:
            The action map.

        """
        actions = ActionMap(input_mode=input_mode)
        for action, scancodes, buttons in (
            (
                PREVIOUS_ACTION,
                (pygame.KSCAN_UP, pygame.KSCAN_LEFT),
                (pygame.CONTROLLER_BUTTON_DPAD_UP, pygame.CONTROLLER_BUTTON_DPAD_LEFT),
            ),
            (
                NEXT_ACTION,
                (pygame.KSCAN_DOWN, pygame.KSCAN_RIGHT),
                (pygame.CONTROLLER_BUTTON_DPAD_DOWN, pygame.CONTROLLER_BUTTON_DPAD_RIGHT),
            ),
            (
                CHOOSE_ACTION,
                (pygame.KSCAN_SPACE, pygame.KSCAN_RETURN),
                (pygame.CONTROLLER_BUTTON_A, pygame.CONTROLLER_BUTTON_START),
            ),
            (
                QUIT_ACTION,
                (pygame.KSCAN_Q,),
                (pygame.CONTROLLER_BUTTON_B, pygame.CONTROLLER_BUTTON_BACK),
            ),
        ):
            for scancode in scancodes:
                actions.bind(action, scancode=scancode)
            for button in buttons:
                actions.bind(action, controller_button=button)
        return actions

    @classmethod
    def args(cls, parser: argparse.ArgumentParser) -> None:
        """Add the command's arguments: ``--game`` (read by the launcher first) and the version.

        Args:
            parser: The argument parser to add arguments to.

        """
        add_game_argument(parser)
        # argparse prints the version and exits as it parses (before the title screen starts)
        parser.add_argument(
            '-v',
            '--version',
            action='version',
            version=f'{cls.NAME} {cls.VERSION}',
            help='print the game version and exit',
        )

    def start_chosen_game(self: Self) -> None:
        """Switch to a new scene of the chosen game, with the run's options, after this frame."""
        choice = GAME_CHOICES[self.selected_index]
        log.info('Starting %s from the title screen', choice.title)
        self.next_scene = choice.scene_class(options=self.options)

    def _select(self: Self, *, step: int) -> None:
        """Choose the game ``step`` cards on (wrapping round), and redraw both cards.

        Args:
            step: -1 for the previous game, 1 for the next.

        """
        self.selected_index = (self.selected_index + step) % len(GAME_CHOICES)
        for index, card in enumerate(self.cards):
            card.select(selected=index == self.selected_index)

    def _on_action_pressed(self: Self, action: str) -> None:
        """Act on a control's press: move the choice, arm starting the game, or quit.

        Args:
            action: The control's action.

        """
        if action == PREVIOUS_ACTION:
            self._select(step=-1)
        elif action == NEXT_ACTION:
            self._select(step=1)
        elif action == CHOOSE_ACTION:
            self._choose_pressed = True
        else:
            # QUIT_ACTION. A QUIT event ends the run, as the games' Q does
            self.scene_manager.quit()

    def _on_action_released(self: Self, action: str | None) -> None:
        """Start the chosen game on the release of a choose control pressed on this screen.

        Args:
            action: The released control's action, or None for an unbound one.

        """
        if action == CHOOSE_ACTION and self._choose_pressed:
            self._choose_pressed = False
            self.start_chosen_game()

    @override
    def on_key_down_event(self: Self, event: KeyEvent) -> None:
        """Act on a bound key's press; pass any other to the base scene (the debug keys).

        Args:
            event: The key down event.

        """
        self.actions.handle_event(event)
        action = self.actions.get_action(event)
        if action is None:
            super().on_key_down_event(event)
        else:
            self._on_action_pressed(action)

    @override
    def on_key_up_event(self: Self, event: KeyEvent) -> None:
        """Act on a key's release (Q quits on its press: not again here; Escape does nothing).

        Args:
            event: The key up event.

        """
        self.actions.handle_event(event)
        self._on_action_released(self.actions.get_action(event))

    @override
    def on_controller_button_down_event(self: Self, event: ControllerButtonEvent) -> None:
        """Act on a mapped gamepad's button press.

        Args:
            event: The controller button down event.

        """
        self._on_button_down(event)

    @override
    def on_controller_button_up_event(self: Self, event: ControllerButtonEvent) -> None:
        """Act on a mapped gamepad's button release.

        Args:
            event: The controller button up event.

        """
        self._on_button_up(event)

    @override
    def on_joy_button_down_event(self: Self, event: JoyButtonEvent) -> None:
        """Act on any other gamepad's button press.

        Args:
            event: The joystick button down event.

        """
        self._on_button_down(event)

    @override
    def on_joy_button_up_event(self: Self, event: JoyButtonEvent) -> None:
        """Act on any other gamepad's button release.

        Args:
            event: The joystick button up event.

        """
        self._on_button_up(event)

    def _on_button_down(self: Self, event: ControllerButtonEvent | JoyButtonEvent) -> None:
        """Act on a gamepad button's press, from either event layer.

        Args:
            event: The button down event.

        """
        self.actions.handle_event(event)
        action = self.actions.get_action(event)
        if action is not None:
            self._on_action_pressed(action)

    def _on_button_up(self: Self, event: ControllerButtonEvent | JoyButtonEvent) -> None:
        """Act on a gamepad button's release, from either event layer.

        Args:
            event: The button up event.

        """
        self.actions.handle_event(event)
        self._on_action_released(self.actions.get_action(event))
