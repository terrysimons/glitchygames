Collision detection and resolution framework.

Provides collision layers, hitbox-aware detection, and push-out resolution. Connects the TOML hitbox system to actual collision checking so sprites with defined hitboxes get accurate collisions.

Usage

collisions = CollisionManager() collisions.add_layer('player') collisions.add_layer('enemies') collisions.check_overlap('player', 'enemies', on_hit)

class CollisionLayer:
    """A named group of sprites that participate in collision.

    Sprites in a layer can be checked against sprites in another
    layer (or the same layer for intra-group collisions).

    Args:
        name: Human-readable name for this layer.

    """

    def __init__(self, name: str) -> None:
        """Initialize the collision layer."""
        self.name = name
        self._sprites: list[Any] = []

    def add(self, sprite: object) -> None:
        """Add a sprite to this layer.

        Args:
            sprite: The sprite to track for collision.

        """
        if sprite not in self._sprites:
            self._sprites.append(sprite)

    def remove(self, sprite: object) -> None:
        """Remove a sprite from this layer.

        Args:
            sprite: The sprite to remove.

        """
        if sprite in self._sprites:
            self._sprites.remove(sprite)

    def clear(self) -> None:
        """Remove all sprites from this layer."""
        self._sprites.clear()

    @property
    def sprites(self) -> list[Any]:
        """All sprites currently in this layer.

        Returns:
            List of sprites in this layer.

        """
        return self._sprites

    def __len__(self) -> int:
        """Number of sprites in this layer.

        Returns:
            Sprite count.

        """
        return len(self._sprites)

class CollisionManager:
    """Manages collision layers and checks between them.

    Provides a clean API for registering sprite groups and running
    collision checks with callbacks. Replaces manual sprite list
    iteration in game code.
    """

    def __init__(self) -> None:
        """Initialize the collision manager."""
        self._layers: dict[str, CollisionLayer] = {}

    def add_layer(self, name: str) -> CollisionLayer:
        """Create and register a named collision layer.

        Args:
            name: Unique name for this layer (e.g., 'player', 'enemies').

        Returns:
            The created CollisionLayer.

        Raises:
            ValueError: If a layer with this name already exists.

        """
        if name in self._layers:
            msg = f"Collision layer '{name}' already exists"
            raise ValueError(msg)
        layer = CollisionLayer(name)
        self._layers[name] = layer
        return layer

    def get_layer(self, name: str) -> CollisionLayer:
        """Get a collision layer by name.

        Args:
            name: The layer name.

        Returns:
            The CollisionLayer.

        Raises:
            KeyError: If no layer with this name exists.

        """
        if name not in self._layers:
            msg = f"No collision layer named '{name}'"
            raise KeyError(msg)
        return self._layers[name]

    def check_overlap(
        self,
        layer_a_name: str,
        layer_b_name: str,
        callback: Callable[[Any, Any], None],
    ) -> None:
        """Check all pairs between two layers, calling callback on collision.

        For each sprite in layer_a that overlaps a sprite in layer_b,
        calls callback(sprite_a, sprite_b). Uses hitbox-aware collision
        rects when available.

        If layer_a and layer_b are the same, checks all unique pairs
        within the layer (no self-collision, no duplicate pairs).

        Each sprite_a's rect is computed once, before its inner loop, so a callback
        that moves sprite_a does not change which later sprite_b it is tested against
        in this pass. Both layers are iterated as snapshots taken at the start, so a
        callback may add or remove sprites (e.g. collect a pickup) without skipping
        pairs; a sprite removed during the pass can still be paired later in it.

        Args:
            layer_a_name: Name of the first layer.
            layer_b_name: Name of the second layer.
            callback: Called with (sprite_a, sprite_b) on each collision.

        """
        layer_a = self.get_layer(layer_a_name)
        layer_b = self.get_layer(layer_b_name)

        same_layer = layer_a_name == layer_b_name
        # Snapshots: callbacks may change the layers while we iterate
        sprites_a = list(layer_a.sprites)
        all_sprites_b = sprites_a if same_layer else list(layer_b.sprites)

        for index_a, sprite_a in enumerate(sprites_a):
            rect_a = get_sprite_collision_rect(sprite_a)

            sprites_b = all_sprites_b[index_a + 1 :] if same_layer else all_sprites_b

            for sprite_b in sprites_b:
                rect_b = get_sprite_collision_rect(sprite_b)
                if check_aabb_overlap(rect_a, rect_b):
                    callback(sprite_a, sprite_b)

    def check_single(
        self,
        sprite: object,
        layer_name: str,
        callback: Callable[[Any, Any], None],
    ) -> None:
        """Check one sprite against all sprites in a layer (iterating a snapshot of it).

        Args:
            sprite: The sprite to check.
            layer_name: Name of the layer to check against.
            callback: Called with (sprite, other) on each collision.

        """
        layer = self.get_layer(layer_name)
        rect = get_sprite_collision_rect(sprite)

        # Snapshot: the callback may change the layer while we iterate
        for other in list(layer.sprites):
            if other is sprite:
                continue
            other_rect = get_sprite_collision_rect(other)
            if check_aabb_overlap(rect, other_rect):
                callback(sprite, other)

    @property
    def layer_names(self) -> list[str]:
        """Names of all registered layers."""
        return list(self._layers.keys())

def compute_push_out(
    mover: pygame.Rect,
    static: pygame.Rect,
    velocity_x: float = 0.0,
) -> tuple[float, float]:
    """Compute the displacement needed to push mover out of static.

    Uses the mover's horizontal velocity to determine push direction.
    If moving right, pushes mover to the left of static; if moving
    left, pushes to the right.

    Args:
        mover: The rect of the moving object.
        static: The rect of the static object.
        velocity_x: Horizontal velocity of the mover (determines push direction).

    Returns:
        Tuple of (dx, dy) displacement to apply to the mover's position.

    """
    if not mover.colliderect(static):
        return (0.0, 0.0)

    # Compute overlap on each axis
    overlap_left = mover.right - static.left
    overlap_right = static.right - mover.left
    overlap_top = mover.bottom - static.top
    overlap_bottom = static.bottom - mover.top

    # Push out along the axis of least penetration
    min_horizontal = min(overlap_left, overlap_right)
    min_vertical = min(overlap_top, overlap_bottom)

    if min_horizontal < min_vertical:
        # Push horizontally
        if velocity_x > 0:
            return (-float(overlap_left), 0.0)
        return (float(overlap_right), 0.0)

    # Push vertically
    if overlap_top < overlap_bottom:
        return (0.0, -float(overlap_top))
    return (0.0, float(overlap_bottom))

def get_collision_rect(
    sprite: RectSprite,
    world_x: float,
    world_y: float,
) -> pygame.Rect:
    """Get the collision rect for a sprite at a world position.

    Uses the sprite's TOML hitbox if available (via get_hitbox_world_rect).
    Otherwise the rect starts at the snapped world position (see
    snapped_world_position) and covers the visible pixels of the sprite's image
    (see opaque_bounds), or the sprite's full rect size when the sprite has no
    image or sets TRIM_TRANSPARENT_HITBOX = False.

    Args:
        sprite: The sprite to get a collision rect for.
        world_x: X position in world space.
        world_y: Y position in world space.

    Returns:
        A pygame.Rect positioned in world coordinates.

    """
    if provides_hitbox_world_rect(sprite):
        return sprite.get_hitbox_world_rect(world_x, world_y)

    snapped_x, snapped_y = snapped_world_position(sprite, world_x, world_y)
    # Looked up by name: a RectSprite needs only a rect (no image: its whole rect collides)
    image = getattr(sprite, 'image', None)
    if image is not None and trims_transparent_hitbox(sprite):
        visible_bounds = opaque_bounds(image)
        return pygame.Rect(
            snapped_x + visible_bounds.x,
            snapped_y + visible_bounds.y,
            visible_bounds.width,
            visible_bounds.height,
        )

    return pygame.Rect(
        snapped_x,
        snapped_y,
        sprite.rect.width,
        sprite.rect.height,
    )

def get_collision_rect_with_margin(
    sprite: RectSprite,
    world_x: float,
    world_y: float,
    margin: int = 0,
) -> pygame.Rect:
    """Get a collision rect with an inset margin for forgiving collisions.

    Shrinks the rect by `margin` pixels on all sides. Useful for
    "forgiveness" in player-enemy collisions where pixel-perfect
    overlap feels unfair.

    Args:
        sprite: The sprite to get a collision rect for.
        world_x: X position in world space.
        world_y: Y position in world space.
        margin: Pixels to shrink on each side (default 0).

    Returns:
        A pygame.Rect shrunk by margin on all sides.

    """
    rect = get_collision_rect(sprite, world_x, world_y)
    if margin > 0:
        rect.inflate_ip(-margin * 2, -margin * 2)
    return rect

bounds #

Visible-pixel bounds of sprite images, used to trim hitboxes to what is drawn.

By default a sprite without a TOML hitbox collides as the smallest rect that contains its image's visible pixels (per-pixel alpha and colorkey are both honoured), not as the whole image. A sprite opts out with a TRIM_TRANSPARENT_HITBOX = False attribute on its class or instance.

Bounds are measured on every call, never cached per surface: several games (Brave Adventurer, Clockwork Keep) create a blank image and redraw each frame's pose into that same surface, so a per-surface cache would keep the blank or first pose. AnimatedSprite frames, whose surfaces do not change, cache their bounds per frame (see SpriteFrame.opaque_bounds).

opaque_bounds(image) #

Return the rect of an image's visible pixels, relative to its top-left corner.

def opaque_bounds(image: pygame.Surface) -> pygame.Rect:
    """Return the rect of an image's visible pixels, relative to its top-left corner.

    Pixels that are fully transparent (alpha 0, or equal to the colorkey) are trimmed. An
    image with no visible pixels returns its full rect, so invisible trigger sprites (and
    images not drawn yet) keep their area. Measured on every call: about 1 us for 20x20,
    8 us for 64x64 and 1 ms for 800x600.

    Args:
        image (pygame.Surface): The sprite image.

    Returns:
        pygame.Rect: A new rect (callers may mutate it).

    """
    visible_bounds = image.get_bounding_rect()
    if visible_bounds.width == 0 or visible_bounds.height == 0:
        return image.get_rect()
    return visible_bounds

def compute_push_out(
    mover: pygame.Rect,
    static: pygame.Rect,
    velocity_x: float = 0.0,
) -> tuple[float, float]:
    """Compute the displacement needed to push mover out of static.

    Uses the mover's horizontal velocity to determine push direction.
    If moving right, pushes mover to the left of static; if moving
    left, pushes to the right.

    Args:
        mover: The rect of the moving object.
        static: The rect of the static object.
        velocity_x: Horizontal velocity of the mover (determines push direction).

    Returns:
        Tuple of (dx, dy) displacement to apply to the mover's position.

    """
    if not mover.colliderect(static):
        return (0.0, 0.0)

    # Compute overlap on each axis
    overlap_left = mover.right - static.left
    overlap_right = static.right - mover.left
    overlap_top = mover.bottom - static.top
    overlap_bottom = static.bottom - mover.top

    # Push out along the axis of least penetration
    min_horizontal = min(overlap_left, overlap_right)
    min_vertical = min(overlap_top, overlap_bottom)

    if min_horizontal < min_vertical:
        # Push horizontally
        if velocity_x > 0:
            return (-float(overlap_left), 0.0)
        return (float(overlap_right), 0.0)

    # Push vertically
    if overlap_top < overlap_bottom:
        return (0.0, -float(overlap_top))
    return (0.0, float(overlap_bottom))

def get_collision_rect(
    sprite: RectSprite,
    world_x: float,
    world_y: float,
) -> pygame.Rect:
    """Get the collision rect for a sprite at a world position.

    Uses the sprite's TOML hitbox if available (via get_hitbox_world_rect).
    Otherwise the rect starts at the snapped world position (see
    snapped_world_position) and covers the visible pixels of the sprite's image
    (see opaque_bounds), or the sprite's full rect size when the sprite has no
    image or sets TRIM_TRANSPARENT_HITBOX = False.

    Args:
        sprite: The sprite to get a collision rect for.
        world_x: X position in world space.
        world_y: Y position in world space.

    Returns:
        A pygame.Rect positioned in world coordinates.

    """
    if provides_hitbox_world_rect(sprite):
        return sprite.get_hitbox_world_rect(world_x, world_y)

    snapped_x, snapped_y = snapped_world_position(sprite, world_x, world_y)
    # Looked up by name: a RectSprite needs only a rect (no image: its whole rect collides)
    image = getattr(sprite, 'image', None)
    if image is not None and trims_transparent_hitbox(sprite):
        visible_bounds = opaque_bounds(image)
        return pygame.Rect(
            snapped_x + visible_bounds.x,
            snapped_y + visible_bounds.y,
            visible_bounds.width,
            visible_bounds.height,
        )

    return pygame.Rect(
        snapped_x,
        snapped_y,
        sprite.rect.width,
        sprite.rect.height,
    )

class CollisionLayer:
    """A named group of sprites that participate in collision.

    Sprites in a layer can be checked against sprites in another
    layer (or the same layer for intra-group collisions).

    Args:
        name: Human-readable name for this layer.

    """

    def __init__(self, name: str) -> None:
        """Initialize the collision layer."""
        self.name = name
        self._sprites: list[Any] = []

    def add(self, sprite: object) -> None:
        """Add a sprite to this layer.

        Args:
            sprite: The sprite to track for collision.

        """
        if sprite not in self._sprites:
            self._sprites.append(sprite)

    def remove(self, sprite: object) -> None:
        """Remove a sprite from this layer.

        Args:
            sprite: The sprite to remove.

        """
        if sprite in self._sprites:
            self._sprites.remove(sprite)

    def clear(self) -> None:
        """Remove all sprites from this layer."""
        self._sprites.clear()

    @property
    def sprites(self) -> list[Any]:
        """All sprites currently in this layer.

        Returns:
            List of sprites in this layer.

        """
        return self._sprites

    def __len__(self) -> int:
        """Number of sprites in this layer.

        Returns:
            Sprite count.

        """
        return len(self._sprites)

class CollisionManager:
    """Manages collision layers and checks between them.

    Provides a clean API for registering sprite groups and running
    collision checks with callbacks. Replaces manual sprite list
    iteration in game code.
    """

    def __init__(self) -> None:
        """Initialize the collision manager."""
        self._layers: dict[str, CollisionLayer] = {}

    def add_layer(self, name: str) -> CollisionLayer:
        """Create and register a named collision layer.

        Args:
            name: Unique name for this layer (e.g., 'player', 'enemies').

        Returns:
            The created CollisionLayer.

        Raises:
            ValueError: If a layer with this name already exists.

        """
        if name in self._layers:
            msg = f"Collision layer '{name}' already exists"
            raise ValueError(msg)
        layer = CollisionLayer(name)
        self._layers[name] = layer
        return layer

    def get_layer(self, name: str) -> CollisionLayer:
        """Get a collision layer by name.

        Args:
            name: The layer name.

        Returns:
            The CollisionLayer.

        Raises:
            KeyError: If no layer with this name exists.

        """
        if name not in self._layers:
            msg = f"No collision layer named '{name}'"
            raise KeyError(msg)
        return self._layers[name]

    def check_overlap(
        self,
        layer_a_name: str,
        layer_b_name: str,
        callback: Callable[[Any, Any], None],
    ) -> None:
        """Check all pairs between two layers, calling callback on collision.

        For each sprite in layer_a that overlaps a sprite in layer_b,
        calls callback(sprite_a, sprite_b). Uses hitbox-aware collision
        rects when available.

        If layer_a and layer_b are the same, checks all unique pairs
        within the layer (no self-collision, no duplicate pairs).

        Each sprite_a's rect is computed once, before its inner loop, so a callback
        that moves sprite_a does not change which later sprite_b it is tested against
        in this pass. Both layers are iterated as snapshots taken at the start, so a
        callback may add or remove sprites (e.g. collect a pickup) without skipping
        pairs; a sprite removed during the pass can still be paired later in it.

        Args:
            layer_a_name: Name of the first layer.
            layer_b_name: Name of the second layer.
            callback: Called with (sprite_a, sprite_b) on each collision.

        """
        layer_a = self.get_layer(layer_a_name)
        layer_b = self.get_layer(layer_b_name)

        same_layer = layer_a_name == layer_b_name
        # Snapshots: callbacks may change the layers while we iterate
        sprites_a = list(layer_a.sprites)
        all_sprites_b = sprites_a if same_layer else list(layer_b.sprites)

        for index_a, sprite_a in enumerate(sprites_a):
            rect_a = get_sprite_collision_rect(sprite_a)

check_overlap(layer_a_name, layer_b_name, callback) #

Check all pairs between two layers, calling callback on collision.

For each sprite in layer_a that overlaps a sprite in layer_b, calls callback(sprite_a, sprite_b). Uses hitbox-aware collision rects when available.

If layer_a and layer_b are the same, checks all unique pairs within the layer (no self-collision, no duplicate pairs).

Each sprite_a's rect is computed once, before its inner loop, so a callback that moves sprite_a does not change which later sprite_b it is tested against in this pass. Both layers are iterated as snapshots taken at the start, so a callback may add or remove sprites (e.g. collect a pickup) without skipping pairs; a sprite removed during the pass can still be paired later in it.

def check_overlap(
    self,
    layer_a_name: str,
    layer_b_name: str,
    callback: Callable[[Any, Any], None],
) -> None:
    """Check all pairs between two layers, calling callback on collision.

    For each sprite in layer_a that overlaps a sprite in layer_b,
    calls callback(sprite_a, sprite_b). Uses hitbox-aware collision
    rects when available.

    If layer_a and layer_b are the same, checks all unique pairs
    within the layer (no self-collision, no duplicate pairs).

    Each sprite_a's rect is computed once, before its inner loop, so a callback
    that moves sprite_a does not change which later sprite_b it is tested against
    in this pass. Both layers are iterated as snapshots taken at the start, so a
    callback may add or remove sprites (e.g. collect a pickup) without skipping
    pairs; a sprite removed during the pass can still be paired later in it.

    Args:
        layer_a_name: Name of the first layer.
        layer_b_name: Name of the second layer.
        callback: Called with (sprite_a, sprite_b) on each collision.

    """

    layer_b = self.get_layer(layer_b_name)

        sprites_b = all_sprites_b[index_a + 1 :] if same_layer else all_sprites_b

def collision_rect_offset(sprite: WorldRectSprite) -> tuple[int, int]:
    """Return the offset of a world sprite's collision rect from its snapped world position.

    Terrain code uses it to place a sprite so its collision rect (not its whole image) is
    flush against a surface: ``world_x = wall_left - (offset_x + rect.width)`` puts the rect's
    right edge on ``wall_left``, and ``world_x = wall_right - offset_x`` puts its left edge on
    ``wall_right``.

    Args:
        sprite (WorldRectSprite): A sprite with world_x and world_y.

    Returns:
        tuple[int, int]: (offset_x, offset_y) of the collision rect's top-left.

    collision_rect = get_collision_rect(sprite, sprite.world_x, sprite.world_y)
    snapped_world_x, snapped_world_y = snapped_world_position(
        sprite, sprite.world_x, sprite.world_y
    )
    return (collision_rect.x - snapped_world_x, collision_rect.y - snapped_world_y)

def get_sprite_collision_rect(sprite: object) -> pygame.Rect:
    """Extract a collision rect from a sprite.

    Handles sprites with world_x/world_y (game objects) and
    sprites with only rect (UI elements).

    Returns:
        A pygame.Rect for collision detection. Sprites with world_x/world_y
        get a rect in world coordinates (see get_collision_rect); sprites with
        only rect get a screen-space rect, trimmed to the visible pixels of
        their image unless they have no image or set
        TRIM_TRANSPARENT_HITBOX = False (then their rect, unchanged); sprites
        with neither get an empty rect.

    """
    import pygame

    if is_world_rect_sprite(sprite):
        return get_collision_rect(sprite, sprite.world_x, sprite.world_y)

    if _has_rect(sprite) and sprite.rect is not None:
        image = getattr(sprite, 'image', None)
        if image is not None and trims_transparent_hitbox(sprite):
            visible_bounds = opaque_bounds(image)
            return pygame.Rect(
                sprite.rect.x + visible_bounds.x,
                sprite.rect.y + visible_bounds.y,
                visible_bounds.width,
                visible_bounds.height,
            )
        return sprite.rect

    return pygame.Rect(0, 0, 0, 0)
